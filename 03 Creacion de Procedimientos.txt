CREATE PROCEDURE [dbo].[usp_TRAMA_BASE_MATERNO_2023_RPT_10_CONSEJERIA]

BEGIN
	SET NOCOUNT ON;
	

	CREATE TABLE #CONSEJ_GEST ([id_cita] [int], id_tipitem varchar(1), cod_item varchar(10) ,[valor_lab] [varchar](6) NULL ) 
	insert into #CONSEJ_GEST
	SELECT distinct  id_cita, id_tipitem, cod_item, valor_lab FROM [TRAMAHIS_DTSG] 
	 where id_cita in
	 (  --- Filtrando los codigos base para el criterio
	   select id_cita--, valor_lab
	   from [dbo].[TRAMAHIS_DTSG]  
	  where   cod_item in ('99401.02')   and id_tipitem in ('D')  and valor_lab='3'
	  AND id_genero='F' and id_tipedad_reg='A'  and edad_reg between 10 and 59  and periodo>='20230101'
	)
	AND  ( cod_item in ('Z3493', 'Z3593') OR (cod_item in ('Z359','Z349') and valor_lab in ('3') and I_ROWNUM_LAB=1 )  ) AND id_tipitem='D'    --- Condición adicional

	CREATE CLUSTERED INDEX IDX_ID_CITA_CONSEJ_GEST ON #CONSEJ_GEST (id_cita)

	----------------------------

	CREATE TABLE #CONSEJ_PUERP ([id_cita] [int], id_tipitem varchar(1), cod_item varchar(10) ,[valor_lab] [varchar](6) NULL ) 
	insert into #CONSEJ_PUERP
	SELECT distinct  id_cita, id_tipitem, cod_item, valor_lab FROM [TRAMAHIS_DTSG] 
	 where id_cita in
	 (  --- Filtrando los codigos base para el criterio
	   select id_cita--, valor_lab
	   from [dbo].[TRAMAHIS_DTSG]  
	  where ( cod_item in ('99401.02')  and id_tipitem in ('D')  and valor_lab='4'  )   
	  AND id_genero='F' and id_tipedad_reg='A'  and edad_reg between 10 and 59 and periodo>='20230101'
	)
	AND  ( cod_item in ('59410','59515')  AND I_ROWNUM_LAB=1 )    --- Condición adicional

	UNION

	SELECT distinct  id_cita, id_tipitem, cod_item, valor_lab FROM [TRAMAHIS_DTSG] 
	 where id_cita in
	 (  --- Filtrando los codigos base para el criterio
	   select id_cita--, valor_lab
	   from [dbo].[TRAMAHIS_DTSG]  
	  where ( cod_item in ('99401.02')  and id_tipitem in ('D')  and valor_lab='5'  )   
	  AND id_genero='F' and id_tipedad_reg='A'  and edad_reg between 10 and 59 and periodo>='20230101'
	)
	AND  ( cod_item in ('59430')  AND I_ROWNUM_LAB=1 )  ;  --- Condición adicional



	CREATE CLUSTERED INDEX IDX_ID_CITA_CONSEJ_PUERP ON #CONSEJ_PUERP (id_cita)




	

    -- Insert statements for procedure here
	 TRUNCATE TABLE TRAMA_BASE_MATERNO_2023_RPT_10_CONSEJERIA_CONSOLIDADO

	IF EXISTS (SELECT * FROM sysobjects WHERE name='TRAMA_BASE_MATERNO_2023_RPT_10_CONSEJERIA_NOMINAL') begin
		drop table TRAMA_BASE_MATERNO_2023_RPT_10_CONSEJERIA_NOMINAL
	END;



	-- GENERACION DE TABLA NOMINAL
	select * into #TEMP from (
		select id_cita FROM #CONSEJ_GEST
		UNION
		select id_cita FROM #CONSEJ_PUERP
	) x

	A.id_genero sexo, A.id_etnia et, A.id_financiador fi, A.id_profesional,
